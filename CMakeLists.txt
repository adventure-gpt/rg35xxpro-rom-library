cmake_minimum_required(VERSION 3.16)
project(rg35xxpro_rom_library VERSION 1.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(DEPS REQUIRED IMPORTED_TARGET sdl2 freetype2 json-glib-1.0 libcurl)
add_executable(rom-library src/main.cpp)
target_compile_options(rom-library PRIVATE -O2 -Wall -Wextra -Wpedantic)
target_link_libraries(rom-library PRIVATE PkgConfig::DEPS pthread)
